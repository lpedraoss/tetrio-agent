import pyautogui
import pyscreeze
from queue import Queue
from core.tetris.predictor_colors import find_colors_tetris_piece
from core.agent.agent import Agent

class TetrioBot():
    def __init__(self) -> None:
        self.agent = Agent()
        self.queue = Queue()
        self.running = False

        # Resolución original donde se capturaron las coordenadas
        self.original_width = 1366
        self.original_height = 768

        # Coordenadas originales
        self.pixels = {
            1: (937, 157),
            2: (946, 257),
            3: (952, 355), 
            4: (943, 451),
            5: (945, 544)
        }

        # Obtener la resolución actual de la pantalla
        screen_width, screen_height = pyautogui.size()

        # Calcular las nuevas coordenadas basadas en la resolución actual
        self.pixel_coords = self.calculate_new_coords(screen_width, screen_height)

        # Initialize board pixel position (also proportionally)
        self.board_pixel_x = int(664 / self.original_width * screen_width)
        self.board_pixel_y = int(66 / self.original_height * screen_height)
        self.color_board = (0, 0, 0)

    def calculate_new_coords(self, screen_width, screen_height):
        """Calcula las nuevas coordenadas basadas en la resolución actual."""
        pixel_coords = {}
        width_ratio = screen_width / self.original_width
        height_ratio = screen_height / self.original_height

        for key, (x, y) in self.pixels.items():
            # Calcular nuevas coordenadas proporcionalmente
            new_x = int(x * width_ratio)
            new_y = int(y * height_ratio)
            pixel_coords[key] = (new_x, new_y)

        return pixel_coords

    def moveDir(self, dir, times):
        if dir != 'center' and times > 0:
            pyautogui.press(dir, presses=times)

    def rota(self, times):
        if times > 0:
            pyautogui.press('x', presses=times)
    
    def moveInBoard(self, dir, times, rotation):
        self.rota(times=rotation)
        self.moveDir(times=times, dir=dir)

    def capturePieceColors(self):
        pieces = []
        try:
            colors = [pyscreeze.pixel(self.pixel_coords[1][0], self.pixel_coords[1][1]),
                      pyscreeze.pixel(self.pixel_coords[2][0], self.pixel_coords[2][1]),
                      pyscreeze.pixel(self.pixel_coords[3][0], self.pixel_coords[3][1]),
                      pyscreeze.pixel(self.pixel_coords[4][0], self.pixel_coords[4][1]),
                      pyscreeze.pixel(self.pixel_coords[5][0], self.pixel_coords[5][1])]
        except KeyError as e:
            print(f"Error: Coordenada no encontrada {e}")
            return pieces
        
        for color in colors:
            piece = find_colors_tetris_piece(color)
            pieces.append(piece)
        return pieces

    def play(self):
        self.running = True
        try:
            initial_pieces = self.capturePieceColors()
            for piece in initial_pieces:
                self.queue.put(piece)
    
            while self.running:
                current_pixel_color = pyscreeze.pixel(self.board_pixel_x, self.board_pixel_y)
                if current_pixel_color != self.color_board:
                    piece = self.queue.get()
                    move = self.agent.startGame(piece=piece)
                    piece, rot, direction, t, move_column = move
                    self.moveInBoard(dir=direction, rotation=rot, times=t)
                    
                    while self.running:
                        new_piece_color = pyscreeze.pixel(self.pixel_coords[5][0], self.pixel_coords[5][1])
                        new_piece = find_colors_tetris_piece(new_piece_color)
                        if new_piece != "board":
                            break
                    self.queue.put(new_piece)
                    pyautogui.press('space')
        
        except KeyError as e:
            print(f"Error: Coordenada no encontrada {e}")
        except Exception as e:
            print(f"Error inesperado: {e}")

    def stop(self):
        self.running = False
